function(dpdf_compile target)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror;-ffp-contract=off>)
  if(DPDF_SANITIZE)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:C>:-fsanitize=address,undefined;-fno-omit-frame-pointer>)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    # Non-PIE test executables avoid intermittent GCC ASan shadow-map startup
    # collisions on high-ASLR Linux/WSL hosts; the shared runtime remains PIC.
    get_target_property(kind ${target} TYPE)
    if(kind STREQUAL "EXECUTABLE")
      target_compile_options(${target} PRIVATE -fno-pie)
      target_link_options(${target} PRIVATE -no-pie)
    endif()
  endif()
endfunction()
# Keep AVX-VNNI optional both at build time and at model initialization.
# Baseline objects must remain executable on AVX2/FMA CPUs without VNNI.
option(DPDF_ENABLE_AVX_VNNI "Build runtime-dispatched AVX-VNNI kernels" ON)
include(CheckCSourceCompiles)
include(CMakePushCheckState)
if(DPDF_ENABLE_AVX_VNNI)
  cmake_push_check_state(RESET)
  set(CMAKE_REQUIRED_FLAGS "-mavx2 -mfma -mavxvnni")
  check_c_source_compiles("#include <immintrin.h>
    int main(void) { __m256i x = _mm256_set1_epi32(__builtin_cpu_supports(\"avxvnni\"));
      x = _mm256_dpbusd_avx_epi32(x, x, x);
      return __builtin_cpu_supports(\"avxvnni\") + _mm256_extract_epi32(x, 0); }"
    DPDF_COMPILER_HAS_AVX_VNNI)
  cmake_pop_check_state()
endif()
foreach(size IN ITEMS 2 8)
  set(model_dir "${CMAKE_CURRENT_SOURCE_DIR}/src/models/dpdfnet${size}_48khz_hr")
  add_library(backend${size} OBJECT src/model_backend.c
    ${model_dir}/generated_model.c src/common/full_ops.c
    src/common/dpdf_dprnn.c src/common/avx2.c src/common/int8.c
    src/common/extended_ops.c src/common/fp16.c
    src/common/qdot64_fixed_avx2.S src/common/qdot4pair_fixed_avx2.S
    ${model_dir}/qaffine64_fused_avx2.S)
  target_include_directories(backend${size} PRIVATE include src src/private)
  target_compile_definitions(backend${size} PRIVATE DPDF_MODEL_SIZE=${size}
    DPDF_X86_DISPATCH DPDF_QASM64 DPDF_QFUSED64 DPDF_QPAIR_FIXED)
  target_compile_options(backend${size} PRIVATE
    -include "${CMAKE_CURRENT_SOURCE_DIR}/src/private/namespace${size}.h"
    $<$<COMPILE_LANGUAGE:C>:-fvisibility=hidden>)
  if(size EQUAL 2)
    target_sources(backend2 PRIVATE ${model_dir}/qaffine256_fused_avx2.S
      ${model_dir}/qaffine4pair64_fused_avx2.S)
    target_compile_definitions(backend2 PRIVATE DPDF_QFUSED256 DPDF_QPAIR_FUSED64)
  endif()
  if(DPDF_ENABLE_AVX_VNNI AND DPDF_COMPILER_HAS_AVX_VNNI)
    target_compile_definitions(backend${size} PRIVATE DPDF_HAS_AVX_VNNI)
    target_sources(backend${size} PRIVATE src/common/int8_vnni.c
      src/common/qdot64_fixed_vnni.S src/common/qdot4pair_fixed_vnni.S
      ${model_dir}/qaffine64_fused_vnni.S)
    if(size EQUAL 2)
      target_sources(backend2 PRIVATE ${model_dir}/qaffine256_fused_vnni.S
        ${model_dir}/qaffine4pair64_fused_vnni.S)
    endif()
  endif()
  dpdf_compile(backend${size})
endforeach()

if(BUILD_TESTING)
  add_executable(public_contract tests/public_contract.c)
  target_link_libraries(public_contract PRIVATE dpdfnet m)
  dpdf_compile(public_contract)
  add_test(NAME public_contract COMMAND public_contract
    "${CMAKE_CURRENT_SOURCE_DIR}/models/dpdfnet2_48khz_hr/weights.f32"
    "${CMAKE_CURRENT_SOURCE_DIR}/models/dpdfnet8_48khz_hr/weights.f32")
  foreach(size IN ITEMS 2 8)
    foreach(test IN ITEMS full_contract extended_contract layout_contract generated_gru_contract
        conv_helpers_contract int8_contract int8_dispatch_contract fused_contract pair_contract)
      set(target "model${size}_${test}")
      add_executable(${target} tests/${test}.c $<TARGET_OBJECTS:backend${size}>)
      target_include_directories(${target} PRIVATE include src src/private)
      target_compile_options(${target} PRIVATE
        -include "${CMAKE_CURRENT_SOURCE_DIR}/src/private/namespace${size}.h" -frounding-math)
      target_compile_definitions(${target} PRIVATE DPDF_X86_DISPATCH DPDF_CONV_PAIR)
      if(DPDF_ENABLE_AVX_VNNI AND DPDF_COMPILER_HAS_AVX_VNNI)
        target_compile_definitions(${target} PRIVATE DPDF_HAS_AVX_VNNI)
      endif()
      target_link_libraries(${target} PRIVATE m)
      if(DPDF_PGO_MODE STREQUAL "GENERATE" AND size EQUAL 8)
        target_link_options(${target} PRIVATE -fprofile-generate=${DPDF_PROFILE_DIR})
      endif()
      dpdf_compile(${target})
      add_test(NAME ${target} COMMAND ${target} "${CMAKE_CURRENT_SOURCE_DIR}/models/dpdfnet${size}_48khz_hr/weights.f32")
    endforeach()
  endforeach()
  foreach(test IN ITEMS wide_contract pair_fused_contract)
    add_executable(model2_${test} tests/${test}.c $<TARGET_OBJECTS:backend2>)
    target_include_directories(model2_${test} PRIVATE include src src/private)
    target_compile_options(model2_${test} PRIVATE
      -include "${CMAKE_CURRENT_SOURCE_DIR}/src/private/namespace2.h" -frounding-math -fno-tree-vectorize)
    target_link_libraries(model2_${test} PRIVATE m)
    dpdf_compile(model2_${test})
    add_test(NAME model2_${test} COMMAND model2_${test})
  endforeach()
  # Reuse the guarded scalar assembly oracles for each VNNI specialization.
  if(DPDF_ENABLE_AVX_VNNI AND DPDF_COMPILER_HAS_AVX_VNNI)
    foreach(size IN ITEMS 2 8)
      set(vnni_contracts fused_contract pair_contract)
      if(size EQUAL 2)
        list(APPEND vnni_contracts wide_contract pair_fused_contract)
      endif()
      foreach(test IN LISTS vnni_contracts)
        set(target "model${size}_${test}_vnni")
        add_executable(${target} tests/${test}.c $<TARGET_OBJECTS:backend${size}>)
        target_include_directories(${target} PRIVATE include src src/private)
        target_compile_options(${target} PRIVATE
          -include "${CMAKE_CURRENT_SOURCE_DIR}/src/private/namespace${size}.h"
          "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/src/private/int8_vnni_aliases.h\""
          -frounding-math -fno-tree-vectorize)
        target_compile_definitions(${target} PRIVATE DPDF_TEST_AVX_VNNI)
        target_link_libraries(${target} PRIVATE m)
        if(DPDF_PGO_MODE STREQUAL "GENERATE" AND size EQUAL 8)
          target_link_options(${target} PRIVATE -fprofile-generate=${DPDF_PROFILE_DIR})
        endif()
        dpdf_compile(${target})
        add_test(NAME ${target} COMMAND ${target})
        set_tests_properties(${target} PROPERTIES SKIP_RETURN_CODE 77)
      endforeach()
    endforeach()
  endif()
  get_property(tests DIRECTORY PROPERTY TESTS)
  set_tests_properties(${tests} PROPERTIES TIMEOUT 120
    ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
endif()
